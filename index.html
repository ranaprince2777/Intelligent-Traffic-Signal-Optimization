🧠 DQN Model Definition (TensorFlow / Keras) 

def build_model(state_size=7, action_size=2):
    model = keras.Sequential([
        layers.Input(shape=(state_size,)),
        layers.Dense(24, activation='relu'),
        layers.Dense(24, activation='relu'),
        layers.Dense(action_size, activation='linear')
    ])
    model.compile(optimizer=keras.optimizers.Adam(learning_rate=0.001), loss='mse')
    return model

# State Space (7 values):
# [Node1_2_EB_0, Node1_2_EB_1, Node1_2_EB_2, Node2_7_SB_0, Node2_7_SB_1, Node2_7_SB_2, current_phase]

# Action Space (2 values):
# 0 = Keep current phase
# 1 = Switch to next phase (subject to MIN_GREEN_STEPS = 5)

⚡ TraCI Control & Feedback Loop 

# Initialize SUMO simulation via TraCI
traci.start([sumo_binary, "-c", "RL.sumocfg", "--step-length", "1.0"])

for step in range(TOTAL_STEPS):
    traci.simulationStep()
    
    # Read E1 Lane Area Detectors
    state = get_state()
    
    # Choose action using epsilon-greedy or DQN policy
    action = get_action_from_policy(state)
    
    # Check minimum green safety constraint
    if action == 1 and (step - last_switch_step >= MIN_GREEN_STEPS):
        switch_to_next_phase()
        last_switch_step = step

    # Collect reward and record ITS metrics
    reward = get_reward(state)
